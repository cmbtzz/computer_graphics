import sys
import os
import importlib.util
from PyQt5.QtWidgets import QApplication
import yaml

# 项目根目录加入搜索路径
BASE_DIR = os.path.dirname(os.path.abspath(__file__))
sys.path.append(BASE_DIR)

from core.viewer_3d import VTKViewerCore
from plugins.base_plugin import Base3DPlugin

def auto_scan_plugins(plugin_dir: str = "plugins") -> dict:
    """自动扫描插件目录，返回 {plugin_cls_name: 插件类}"""
    plugin_map = {}
    plugins_path = os.path.join(BASE_DIR, plugin_dir)

    for file in os.listdir(plugins_path):
        if not file.endswith(".py") or file.startswith("base_"):
            continue
        module_name = file[:-3]
        file_path = os.path.join(plugins_path, file)

        # 动态导入插件模块
        spec = importlib.util.spec_from_file_location(module_name, file_path)
        mod = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(mod)

        # 筛选合法插件子类
        for attr_name in dir(mod):
            attr = getattr(mod, attr_name)
            if (
                isinstance(attr, type)
                and issubclass(attr, Base3DPlugin)
                and attr is not Base3DPlugin
                and getattr(attr, "plugin_cls_name", "")
            ):
                key = attr.plugin_cls_name
                plugin_map[key] = attr
    return plugin_map

if __name__ == "__main__":
    app = QApplication(sys.argv)
    viewer = VTKViewerCore()

    # 自动扫描全部插件
    plugin_mapping = auto_scan_plugins()

    # 读取YAML配置文件
    config_path = os.path.join(BASE_DIR, "plugin_config.yaml")
    active_list = []
    try:
        with open(config_path, "r", encoding="utf-8") as f:
            cfg = yaml.safe_load(f)
        active_list = cfg.get("active_plugins", [])
    except FileNotFoundError:
        print("提示：未找到 plugin_config.yaml，自动加载所有可用插件")
        active_list = list(plugin_mapping.keys())
    except Exception as e:
        print(f"YAML配置读取异常: {e}，自动加载所有插件")
        active_list = list(plugin_mapping.keys())

    # 按配置注册插件
    for plug_key in active_list:
        if plug_key in plugin_mapping:
            viewer.register_plugin(plugin_mapping[plug_key])
            print(f"已加载插件: {plug_key}")
        else:
            print(f"警告：不存在标识为 [{plug_key}] 的插件，跳过")

    viewer.show()
    sys.exit(app.exec_())

'''
框架内核：基础3D查看器底座，仅保留纯基础功能，所有扩展通过插件实现

import sys
import os
import vtk
from vtk.qt.QVTKRenderWindowInteractor import QVTKRenderWindowInteractor
from PyQt5.QtWidgets import (QApplication, QMainWindow, QVBoxLayout, QHBoxLayout, QWidget,
                             QFileDialog, QPushButton, QTextEdit, QTreeView, QFileSystemModel, QSplitter,
                             QMenuBar, QMenu, QAction, QFrame)
from PyQt5.QtCore import QDir, Qt

class ViewerContext:
    """全局上下文，插件共享状态，避免各自重复存储变量"""
    def __init__(self):
        self.main_window = None
        self.vtk_renderer = None
        self.vtk_widget = None
        self.info_display = None
        self.right_extension_panel = None  # 插件面板挂载容器
        self.current_obj_path = None
        self.current_actor = None
        self.current_polydata = None
        self.plugins = []  # 已加载插件实例列表

class VTKViewerCore(QMainWindow):
    def __init__(self):
        super().__init__()
        self.setWindowTitle("3D Model Viewer Framework")
        self.setGeometry(100, 100, 1600, 800)

        # 全局上下文
        self.ctx = ViewerContext()
        self.ctx.main_window = self

        # 插件管理器
        self.plugin_instances = []

        self.create_menu()
        self.build_base_layout()
        self.init_vtk()

    def register_plugin(self, plugin_cls):
        """外部注册插件，入口main.py调用"""
        plugin = plugin_cls(self.ctx)
        self.plugin_instances.append(plugin)
        self.ctx.plugins.append(plugin)
        # 创建插件UI并挂载到右侧扩展面板
        panel = plugin.create_ui()
        if panel:
            self.ctx.right_extension_layout.addWidget(panel)

    def build_base_layout(self):
        """基础布局：文件树 | VTK窗口 | 右侧插件垂直容器"""
        central_widget = QWidget()
        self.setCentralWidget(central_widget)
        main_v_layout = QVBoxLayout(central_widget)
        main_v_layout.setSpacing(4)

        top_h_splitter = QSplitter(Qt.Horizontal)
        main_v_layout.addWidget(top_h_splitter, stretch=8)

        # 底部信息栏
        self.ctx.info_display = QTextEdit()
        self.ctx.info_display.setReadOnly(True)
        self.ctx.info_display.setMaximumHeight(160)
        main_v_layout.addWidget(self.ctx.info_display, stretch=1)

        # 左侧文件树
        self.tree_view = QTreeView()
        self.file_model = QFileSystemModel()
        self.file_model.setFilter(QDir.NoDotAndDotDot | QDir.AllDirs | QDir.Files)
        self.file_model.setNameFilters(["*.obj"])
        self.file_model.setNameFilterDisables(False)
        self.tree_view.setModel(self.file_model)
        self.tree_view.clicked.connect(self.on_tree_item_clicked)
        for col in range(1, self.file_model.columnCount()):
            self.tree_view.hideColumn(col)
        self.file_model.setRootPath("")
        self.tree_view.setRootIndex(self.file_model.index(""))
        top_h_splitter.addWidget(self.tree_view)

        # 中间VTK窗口
        self.ctx.vtk_widget = QVTKRenderWindowInteractor(self)
        self.ctx.vtk_renderer = vtk.vtkRenderer()
        self.ctx.vtk_widget.GetRenderWindow().AddRenderer(self.ctx.vtk_renderer)
        top_h_splitter.addWidget(self.ctx.vtk_widget)

        # 右侧：插件统一挂载容器
        right_container = QWidget()
        self.ctx.right_extension_layout = QVBoxLayout(right_container)
        self.ctx.right_extension_layout.setSpacing(8)
        self.ctx.right_extension_layout.setContentsMargins(0,0,0,0)
        self.ctx.right_extension_panel = right_container
        top_h_splitter.addWidget(right_container)

        top_h_splitter.setSizes([int(0.3*self.width()), int(0.5*self.width()), int(0.2*self.width())])
        top_h_splitter.setStretchFactor(1, 6)
        top_h_splitter.setStretchFactor(2, 4)

    def init_vtk(self):
        self.ctx.vtk_widget.Initialize()
        self.ctx.vtk_widget.Start()

    def create_menu(self):
        menubar = self.menuBar()
        file_menu = QMenu("File", self)
        open_folder_action = QAction("Open Folder", self)
        open_folder_action.triggered.connect(self.open_folder)
        exit_action = QAction("Exit", self)
        exit_action.triggered.connect(self.close)
        file_menu.addAction(open_folder_action)
        file_menu.addAction(exit_action)
        menubar.addMenu(file_menu)

    def open_folder(self):
        folder = QFileDialog.getExistingDirectory(self, "Open Folder", "")
        if folder:
            self.file_model.setRootPath(folder)
            self.tree_view.setRootIndex(self.file_model.index(folder))

    def on_tree_item_clicked(self, index):
        file_path = self.file_model.filePath(index)
        if os.path.isfile(file_path) and file_path.endswith(".obj"):
            self.load_model(file_path)

    def load_model(self, file_name):
        """基础OBJ加载逻辑，纯白模型，加载完成分发插件钩子"""
        vtk.vtkObject.GlobalWarningDisplayOff()
        reader = vtk.vtkOBJReader()
        reader.SetFileName(file_name)
        reader.Update()
        polydata = reader.GetOutput()
        num_points = polydata.GetPoints().GetNumberOfPoints()
        num_faces = polydata.GetNumberOfCells()

        # 纯白顶点色
        color_array = vtk.vtkUnsignedCharArray()
        color_array.SetNumberOfComponents(3)
        color_array.SetName("VertexColors")
        for i in range(num_points):
            color_array.InsertNextTuple3(255,255,255)
        polydata.GetPointData().SetScalars(color_array)

        mapper = vtk.vtkPolyDataMapper()
        mapper.SetInputData(polydata)
        actor = vtk.vtkActor()
        actor.SetMapper(mapper)

        # 更新全局上下文
        self.ctx.current_obj_path = file_name
        self.ctx.current_actor = actor
        self.ctx.current_polydata = polydata

        self.ctx.vtk_renderer.RemoveAllViewProps()
        self.ctx.vtk_renderer.AddActor(actor)
        self.add_axes(polydata)
        self.ctx.vtk_renderer.ResetCamera()
        self.ctx.vtk_widget.GetRenderWindow().Render()

        self.update_info(file_name, num_points, num_faces)

        # 分发插件钩子：模型加载完成
        for plugin in self.plugin_instances:
            plugin.on_model_loaded(file_name, actor, polydata)
        return True

    def add_axes(self, polydata):
        """基础坐标轴，全局统一"""
        centroid = polydata.GetCenter()
        axes_actor = vtk.vtkAxesActor()
        axes_actor.SetTotalLength(15,15,15)
        x_c = axes_actor.GetXAxisCaptionActor2D()
        y_c = axes_actor.GetYAxisCaptionActor2D()
        z_c = axes_actor.GetZAxisCaptionActor2D()
        x_c.GetCaptionTextProperty().SetColor(1,0,0)
        y_c.GetCaptionTextProperty().SetColor(0,1,0)
        z_c.GetCaptionTextProperty().SetColor(0,0,1)
        x_c.GetCaptionTextProperty().SetFontSize(10)
        y_c.GetCaptionTextProperty().SetFontSize(10)
        z_c.GetCaptionTextProperty().SetFontSize(10)
        offset = 24
        x_c.SetAttachmentPoint(centroid[0]+offset, centroid[1], centroid[2])
        y_c.SetAttachmentPoint(centroid[0], centroid[1]+offset, centroid[2])
        z_c.SetAttachmentPoint(centroid[0], centroid[1], centroid[2]+offset)
        trans = vtk.vtkTransform()
        trans.Translate(centroid[0], centroid[1], centroid[2]+10)
        axes_actor.SetUserTransform(trans)
        self.ctx.vtk_renderer.AddActor(axes_actor)

    def update_info(self, obj_file, pts, faces):
        text = f"OBJ File: {obj_file}\nVertices: {pts}\nFaces: {faces}\n"
        self.ctx.info_display.setPlainText(text)

    def resizeEvent(self, event):
        super().resizeEvent(event)
        # 窗口缩放分发插件钩子
        for plugin in self.plugin_instances:
            plugin.on_window_resize()

    def closeEvent(self, event):
        # 程序关闭释放插件资源
        for plugin in self.plugin_instances:
            plugin.on_destroy()
        event.accept()

from PyQt5.QtWidgets import QWidget, QVBoxLayout, QHBoxLayout, QCheckBox, QFrame
from PyQt5.QtCore import Qt
import vtk
from plugins.base_plugin import Base3DPlugin

class LightPlugin(Base3DPlugin):
    plugin_cls_name = "light_control"
    plugin_name = "light_control"
    plugin_desc = "光照材质调节插件"
    
    def __init__(self, viewer_ctx):
        super().__init__(viewer_ctx)
        self.light_global_enable = True
        self.light_ambient = True
        self.light_diffuse = True
        self.light_specular = True
        self.scene_light = None
        self.chk_global = None
        self.chk_ambient = None
        self.chk_diffuse = None
        self.chk_specular = None

    def create_ui(self):
        self.plugin_panel = QFrame()
        self.plugin_panel.setFrameShape(QFrame.StyledPanel)
        layout = QVBoxLayout(self.plugin_panel)
        layout.setSpacing(6)
        layout.setContentsMargins(10,10,10,10)

        # 全局光照开关
        self.chk_global = QCheckBox("启用光照")
        self.chk_global.setChecked(self.light_global_enable)
        self.chk_global.stateChanged.connect(self.on_global_toggle)
        layout.addWidget(self.chk_global)

        h_layout = QHBoxLayout()
        self.chk_ambient = QCheckBox("环境光")
        self.chk_ambient.setChecked(self.light_ambient)
        self.chk_ambient.stateChanged.connect(self.update_material)

        self.chk_diffuse = QCheckBox("漫反射")
        self.chk_diffuse.setChecked(self.light_diffuse)
        self.chk_diffuse.stateChanged.connect(self.update_material)

        self.chk_specular = QCheckBox("镜面高光")
        self.chk_specular.setChecked(self.light_specular)
        self.chk_specular.stateChanged.connect(self.update_material)

        h_layout.addWidget(self.chk_ambient)
        h_layout.addWidget(self.chk_diffuse)
        h_layout.addWidget(self.chk_specular)
        layout.addLayout(h_layout)

        # 创建场景光源
        self.create_light()
        self.set_sub_enable(True)
        return self.plugin_panel

    def create_light(self):
        self.scene_light = vtk.vtkLight()
        self.scene_light.SetPosition(-100,100,120)
        self.scene_light.SetFocalPoint(0,0,0)
        self.scene_light.SetIntensity(0.8)
        self.scene_light.SetColor(1,1,1)
        self.scene_light.SetLightTypeToSceneLight()
        self.ctx.vtk_renderer.AddLight(self.scene_light)

    def set_sub_enable(self, enable):
        self.chk_ambient.setEnabled(enable)
        self.chk_diffuse.setEnabled(enable)
        self.chk_specular.setEnabled(enable)

    def on_global_toggle(self, state):
        self.light_global_enable = (state == Qt.Checked)
        self.set_sub_enable(self.light_global_enable)
        self.scene_light.SetSwitch(self.light_global_enable)
        self.update_material()
        self.ctx.vtk_widget.GetRenderWindow().Render()

    def update_material(self):
        if not self.ctx.current_actor:
            return
        mat = self.ctx.current_actor.GetProperty()
        if not self.light_global_enable:
            mat.SetAmbient(0.0)
            mat.SetDiffuse(0.0)
            mat.SetSpecular(0.0)
        else:
            mat.SetAmbient(0.2 if self.chk_ambient.isChecked() else 0.0)
            mat.SetDiffuse(0.8 if self.chk_diffuse.isChecked() else 0.0)
            mat.SetSpecular(0.4 if self.chk_specular.isChecked() else 0.0)
        self.ctx.vtk_widget.GetRenderWindow().Render()

    def on_model_loaded(self, obj_path, actor, polydata):
        # 新模型加载自动同步光照参数
        self.update_material()

import os
from PyQt5.QtWidgets import QWidget, QScrollArea, QGridLayout, QPushButton, QVBoxLayout
from PyQt5.QtCore import Qt, QSize, QTimer
from PyQt5.QtGui import QPixmap, QIcon
import vtk
from plugins.base_plugin import Base3DPlugin

class TexturePlugin(Base3DPlugin):
    # json配置里写这个字符串
    plugin_cls_name = "texture_mapping"
    plugin_name = "texture_mapping"
    plugin_desc = "纹理贴图功能插件"

    def __init__(self, viewer_ctx):
        super().__init__(viewer_ctx)
        self.selected_texture_path = None
        self.selected_texture_btn = None
        self.texture_grid = None
        self.texture_scroll = None
        self.timer = QTimer()
        self.timer.setSingleShot(True)
        self.timer.timeout.connect(self.load_texture_images)

    def create_ui(self):
        """构建纹理缩略图面板，挂载到右侧扩展区"""
        self.plugin_panel = QWidget()
        layout = QVBoxLayout(self.plugin_panel)
        layout.setContentsMargins(0,0,0,0)

        self.texture_scroll = QScrollArea()
        self.texture_scroll.setWidgetResizable(True)
        self.texture_scroll.setVerticalScrollBarPolicy(Qt.ScrollBarAsNeeded)
        self.texture_scroll.setHorizontalScrollBarPolicy(Qt.ScrollBarAlwaysOff)

        container = QWidget()
        self.texture_grid = QGridLayout(container)
        self.texture_grid.setSpacing(6)
        self.texture_grid.setContentsMargins(15,15,15,15)
        for c in range(4):
            self.texture_grid.setColumnStretch(c, 1)
        self.texture_scroll.setWidget(container)
        layout.addWidget(self.texture_scroll)

        # 延迟加载纹理图
        QTimer.singleShot(100, self.load_texture_images)
        return self.plugin_panel

    def on_window_resize(self):
        # 窗口缩放延迟刷新纹理缩略图
        self.timer.start(50)

    def on_model_loaded(self, obj_path, actor, polydata):
        # 加载模型后自动应用已选纹理
        if self.selected_texture_path:
            self.apply_texture(self.selected_texture_path)

    def on_texture_btn_click(self, btn, tex_path):
        if self.selected_texture_btn:
            self.selected_texture_btn.setStyleSheet("")
        btn.setStyleSheet("border:3px solid #ff4400;")
        self.selected_texture_btn = btn
        self.selected_texture_path = tex_path
        self.ctx.info_display.append(f"\n选中纹理：{tex_path}")
        if self.ctx.current_actor:
            self.apply_texture(tex_path)

    def apply_texture(self, tex_path):
        # 纹理映射逻辑，完全隔离在插件
        if tex_path.lower().endswith((".jpg", ".jpeg")):
            reader = vtk.vtkJPEGReader()
        else:
            reader = vtk.vtkPNGReader()
        reader.SetFileName(tex_path)
        reader.Update()
        texture = vtk.vtkTexture()
        texture.SetInputData(reader.GetOutput())
        texture.RepeatOn()
        texture.InterpolateOn()

        tex_gen = vtk.vtkTextureMapToCylinder()
        tex_gen.SetInputData(self.ctx.current_actor.GetMapper().GetInput())
        tex_gen.Update()

        mapper = vtk.vtkPolyDataMapper()
        mapper.SetInputData(tex_gen.GetOutput())
        self.ctx.current_actor.SetMapper(mapper)
        self.ctx.current_actor.SetTexture(texture)
        self.ctx.vtk_widget.GetRenderWindow().Render()
        self.ctx.info_display.append(f"纹理已绑定模型：{os.path.basename(self.ctx.current_obj_path)}")

    def load_texture_images(self):
        max_r, max_c = 5, 4
        total = max_r * max_c
        # 清空旧控件
        while self.texture_grid.count():
            item = self.texture_grid.takeAt(0)
            w = item.widget()
            if w: w.deleteLater()

        view_w = self.texture_scroll.viewport().width()
        if view_w <= 10: view_w = 200
        spacing = self.texture_grid.spacing()
        mg = self.texture_grid.contentsMargins()
        avail = view_w - mg.left() - mg.right() - (max_c-1)*spacing
        thumb_size = max(64, int(avail / max_c))

        tex_dir = "../data/texture"
        img_list = []
        if os.path.exists(tex_dir):
            for f in sorted(os.listdir(tex_dir)):
                if f.lower().endswith((".png",".jpg",".jpeg")):
                    img_list.append(os.path.join(tex_dir, f))

        slot = 0
        for r in range(max_r):
            for c in range(max_c):
                if slot < len(img_list):
                    path = img_list[slot]
                    pix = QPixmap(path).scaled(thumb_size, thumb_size, Qt.KeepAspectRatio, Qt.SmoothTransformation)
                    btn = QPushButton()
                    btn.setFixedSize(thumb_size, thumb_size)
                    btn.setIcon(QIcon(pix))
                    btn.setIconSize(QSize(thumb_size-8, thumb_size-8))
                    btn.clicked.connect(lambda ch, b=btn, p=path: self.on_texture_btn_click(b,p))
                    self.texture_grid.addWidget(btn, r, c)
                else:
                    empty = QPushButton()
                    empty.setFixedSize(thumb_size, thumb_size)
                    empty.setEnabled(False)
                    empty.setStyleSheet("background:#eee;border:1px solid #ccc;")
                    self.texture_grid.addWidget(empty, r, c)
                slot += 1
